## What We're Building

A chat application with conversation with GPT models, message history in Room database, multiple conversation threads, streaming responses, API key management, offline mode, and token usage tracking.

## Implementation Highlights

**Room Database**:
```kotlin
@Entity(tableName = "messages")
data class Message(
    @PrimaryKey val id: String = UUID.randomUUID().toString(),
    val conversationId: String,
    val role: String,
    val content: String,
    val createdAt: Long = System.currentTimeMillis()
)

@Dao
interface MessageDao {
    @Query("SELECT * FROM messages WHERE conversationId = :conversationId ORDER BY createdAt ASC")
    suspend fun getMessages(conversationId: String): List<Message>

    @Insert
    suspend fun insert(message: Message)
}

@Database(entities = [Message::class, Conversation::class], version = 1)
abstract class ChatDatabase : RoomDatabase() {
    abstract fun messageDao(): MessageDao
    abstract fun conversationDao(): ConversationDao
}
```

**ViewModel with Coroutines**:
```kotlin
class ChatViewModel(
    private val conversationId: String,
    private val openAIService: OpenAIService,
    private val database: ChatDatabase
) : ViewModel() {

    private val _messages = MutableStateFlow<List<Message>>(emptyList())
    val messages: StateFlow<List<Message>> = _messages.asStateFlow()

    init {
        loadMessages()
    }

    private fun loadMessages() {
        viewModelScope.launch {
            _messages.value = database.messageDao().getMessages(conversationId)
        }
    }

    fun sendMessage(content: String) {
        viewModelScope.launch {
            val userMessage = Message(
                conversationId = conversationId,
                role = "user",
                content = content
            )

            database.messageDao().insert(userMessage)
            loadMessages()

            try {
                val chatMessages = _messages.value.map {
                    ChatMessage(role = it.role, content = it.content)
                }

                val response = openAIService.sendMessage(chatMessages)

                val assistantMessage = Message(
                    conversationId = conversationId,
                    role = "assistant",
                    content = response.content
                )

                database.messageDao().insert(assistantMessage)
                loadMessages()
            } catch (e: Exception) {
                Log.e("ChatViewModel", "Error: ${e.message}")
            }
        }
    }
}
```

**Compose UI**:
```kotlin
@Composable
fun ChatScreen(viewModel: ChatViewModel) {
    val messages by viewModel.messages.collectAsState()
    var inputText by remember { mutableStateOf("") }

    Column(Modifier.fillMaxSize()) {
        LazyColumn(
            modifier = Modifier.weight(1f),
            reverseLayout = true
        ) {
            items(messages.reversed()) { message ->
                MessageBubble(message)
            }
        }

        Row(Modifier.padding(8.dp)) {
            TextField(
                value = inputText,
                onValueChange = { inputText = it },
                modifier = Modifier.weight(1f)
            )

            Button(onClick = {
                viewModel.sendMessage(inputText)
                inputText = ""
            }) {
                Text("Send")
            }
        }
    }
}
```

## Common Pitfalls

1. **Main Thread Blocking**: Never call suspend functions from UI thread directly
2. **API Key Security**: Use BuildConfig or encrypted SharedPreferences, not hardcoded strings
3. **Memory Leaks**: ViewModels survive configuration changes, Activities don't
4. **Token Limits**: GPT-4 has context limits. Truncate old messages.

## Performance & Costs

Latency: ~500ms per request. Room writes: <1ms. GPT-4o-mini: ~$0.01 per 100 messages.
