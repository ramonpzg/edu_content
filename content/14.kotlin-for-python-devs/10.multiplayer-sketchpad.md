data class LWWRegister<T>(
    var value: T,
    var timestamp: Long,
    var replicaId: String
) {
    fun merge(other: LWWRegister<T>) {
        if (other.timestamp > timestamp ||
            (other.timestamp == timestamp && other.replicaId > replicaId)) {
            this.value = other.value
            this.timestamp = other.timestamp
            this.replicaId = other.replicaId
        }
    }
}

class ORSet<T> {
    private val elements = mutableMapOf<T, MutableSet<String>>()

    fun add(element: T) {
        val tag = UUID.randomUUID().toString()
        elements.getOrPut(element) { mutableSetOf() }.add(tag)
    }

    fun remove(element: T) {
        elements[element]?.clear()
    }

    fun contains(element: T) = elements[element]?.isNotEmpty() == true

    fun merge(other: ORSet<T>) {
        other.elements.forEach { (element, tags) ->
            elements.getOrPut(element) { mutableSetOf() }.addAll(tags)
        }
    }
}

**Stroke CRDT**:
```kotlin
data class Stroke(
    val id: String = UUID.randomUUID().toString(),
    val points: List<StrokePoint>,
    val color: Int,
    val width: Float,
    val ownerId: String,
    val createdAt: Long = System.currentTimeMillis()
) {
    override fun equals(other: Any?) = (other as? Stroke)?.id == id
    override fun hashCode() = id.hashCode()
}

data class StrokePoint(
    val x: Float,
    val y: Float,
    val pressure: Float,
    val timestamp: Long
)

class StrokeCRDT {
    private val strokes = ORSet<Stroke>()
    private val deletedStrokes = mutableSetOf<String>()

    fun addStroke(stroke: Stroke) {
        strokes.add(stroke)
    }

    fun deleteStroke(id: String) {
        deletedStrokes.add(id)
    }

    fun visibleStrokes() = strokes.elements
        .filter { it.value.isNotEmpty() }
        .map { it.key }
        .filter { !deletedStrokes.contains(it.id) }

    fun merge(other: StrokeCRDT) {
        strokes.merge(other.strokes)
        deletedStrokes.addAll(other.deletedStrokes)
    }
}
```

**Sync Engine**:
```kotlin
class SyncEngine(private val localUserId: String) {
    private val _state = MutableStateFlow(StrokeCRDT())
    val state: StateFlow<StrokeCRDT> = _state.asStateFlow()

    private val webSocketClient = WebSocketClient("ws://localhost:8080/sketch")

    fun connect() {
        webSocketClient.connect(object : WebSocketListener() {
            override fun onMessage(webSocket: WebSocket, text: String) {
                val remote = Json.decodeFromString<StrokeCRDT>(text)
                mergeRemoteState(remote)
            }
        })
    }

    fun addStroke(stroke: Stroke) {
        _state.value.addStroke(stroke)
        broadcastState()
    }

    private fun mergeRemoteState(remote: StrokeCRDT) {
        val merged = StrokeCRDT()
        merged.merge(_state.value)
        merged.merge(remote)
        _state.value = merged
    }

    private fun broadcastState() {
        val json = Json.encodeToString(_state.value)
        webSocketClient.send(json)
    }
}
```

**Canvas Drawing**:
```kotlin
@Composable
fun DrawingCanvas(syncEngine: SyncEngine) {
    val state by syncEngine.state.collectAsState()
    var currentStroke by remember { mutableStateOf<Stroke?>(null) }

    Canvas(
        modifier = Modifier
            .fillMaxSize()
            .pointerInput(Unit) {
                detectDragGestures(
                    onDragStart = { offset ->
                        currentStroke = Stroke(
                            points = listOf(StrokePoint(offset.x, offset.y, 1f, System.currentTimeMillis())),
                            color = Color.Black.toArgb(),
                            width = 5f,
                            ownerId = syncEngine.localUserId
                        )
                    },
                    onDrag = { change, _ ->
                        currentStroke = currentStroke?.copy(
                            points = currentStroke!!.points + StrokePoint(
                                change.position.x,
                                change.position.y,
                                1f,
                                System.currentTimeMillis()
                            )
                        )
                    },
                    onDragEnd = {
                        currentStroke?.let { syncEngine.addStroke(it) }
                        currentStroke = null
                    }
                )
            }
    ) {
        // Draw all strokes
        state.visibleStrokes().forEach { stroke ->
            val path = Path()
            stroke.points.forEachIndexed { index, point ->
                if (index == 0) path.moveTo(point.x, point.y)
                else path.lineTo(point.x, point.y)
            }
            drawPath(path, Color(stroke.color), style = Stroke(width = stroke.width))
        }

        // Draw current stroke
        currentStroke?.let { stroke ->
            val path = Path()
            stroke.points.forEachIndexed { index, point ->
                if (index == 0) path.moveTo(point.x, point.y)
                else path.lineTo(point.x, point.y)
            }
            drawPath(path, Color(stroke.color), style = Stroke(width = stroke.width))
        }
    }
}
```

## Common Pitfalls

1. **CRDT Monotonicity**: Using wall-clock time breaks with clock skew. Use logical clocks.
2. **Unbounded Growth**: CRDTs never delete data. Implement garbage collection for old tombstones.
3. **Network Partitions**: On reconnect, full state transfer can be huge. Use delta-based sync.

## Performance Notes

CRDT merge: O(n) where n = strokes. 1000 strokes: ~0.1ms. Network bandwidth with full state: 500KB per sync. Use deltas for efficiency.
