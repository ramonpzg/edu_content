# 3D Data Visualizations with Three.js

Build interactive 3D visualizations for educational content. Render animated charts, explorable diagrams, and GPU-accelerated graphics at 60fps in the browser.

## What We're Building

Three.js (r181) visualizations:

- 3D bar chart for lesson completion stats
- Interactive molecule viewer (educational chemistry)
- Animated sorting algorithm visualizer
- 3D network graph (lesson dependencies)
- Orbit controls for exploration

**Performance target**: 60fps with 10,000+ objects. GPU does the heavy lifting.

## Basic Three.js Setup

```html
<div id="chart" style="width: 800px; height: 600px;"></div>

<script type="module">
import * as THREE from 'three';
import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

// Scene, Camera, Renderer
const scene = new THREE.Scene();
scene.background = new THREE.Color(0x0d0d0d);

const camera = new THREE.PerspectiveCamera(
  75,                                    // Field of view
  800 / 600,                             // Aspect ratio
  0.1,                                   // Near clipping plane
  1000                                   // Far clipping plane
);
camera.position.set(5, 5, 5);
camera.lookAt(0, 0, 0);

const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(800, 600);
document.getElementById('chart').appendChild(renderer.domElement);

// Orbit controls (mouse drag to rotate)
const controls = new OrbitControls(camera, renderer.domElement);
controls.enableDamping = true;
controls.dampingFactor = 0.05;

// Lighting
const ambientLight = new THREE.AmbientLight(0xffffff, 0.5);
scene.add(ambientLight);

const directionalLight = new THREE.DirectionalLight(0xffffff, 0.8);
directionalLight.position.set(10, 10, 10);
scene.add(directionalLight);

// Animation loop
function animate() {
  requestAnimationFrame(animate);
  controls.update();
  renderer.render(scene, camera);
}

animate();
</script>
```

This creates an empty 3D scene with camera controls.

## 3D Bar Chart

Visualize lesson completion statistics:

```javascript
// Fetch data from API
const response = await fetch('/api/stats/lessons');
const lessons = await response.json();

// Create bar chart
const barWidth = 0.6;
const barSpacing = 1.2;

lessons.forEach((lesson, i) => {
  const height = lesson.completion_rate * 5;  // Scale to 0-5 units

  // Bar geometry
  const geometry = new THREE.BoxGeometry(barWidth, height, barWidth);
  const material = new THREE.MeshStandardMaterial({
    color: new THREE.Color().setHSL(lesson.completion_rate, 0.7, 0.5),
    metalness: 0.3,
    roughness: 0.7,
  });

  const bar = new THREE.Mesh(geometry, material);
  bar.position.set(
    (i - lessons.length / 2) * barSpacing,  // X: spread across scene
    height / 2,                              // Y: bottom at y=0
    0                                         // Z: center
  );

  bar.userData = { lesson: lesson.title, rate: lesson.completion_rate };
  scene.add(bar);

  // Label (using TextGeometry or CSS3DRenderer)
  const label = createTextLabel(lesson.title.slice(0, 8));
  label.position.set(bar.position.x, -0.5, 0);
  scene.add(label);
});

// Raycaster for hover interaction
const raycaster = new THREE.Raycaster();
const mouse = new THREE.Vector2();

renderer.domElement.addEventListener('mousemove', (event) => {
  const rect = renderer.domElement.getBoundingClientRect();
  mouse.x = ((event.clientX - rect.left) / rect.width) * 2 - 1;
  mouse.y = -((event.clientY - rect.top) / rect.height) * 2 + 1;

  raycaster.setFromCamera(mouse, camera);
  const intersects = raycaster.intersectObjects(scene.children);

  if (intersects.length > 0 && intersects[0].object.userData.lesson) {
    const data = intersects[0].object.userData;
    showTooltip(data.lesson, data.rate);
  } else {
    hideTooltip();
  }
});
```

## Animated Sorting Visualizer

Visualize sorting algorithms in 3D:

```javascript
// Generate random array
const array = Array.from({ length: 50 }, () => Math.random());
const bars = [];

// Create bars
array.forEach((value, i) => {
  const height = value * 10;
  const geometry = new THREE.BoxGeometry(0.15, height, 0.15);
  const material = new THREE.MeshStandardMaterial({ color: 0x4a9eff });

  const bar = new THREE.Mesh(geometry, material);
  bar.position.set((i - 25) * 0.2, height / 2, 0);

  bars.push(bar);
  scene.add(bar);
});

// Bubble sort with visualization
async function bubbleSort() {
  for (let i = 0; i < array.length; i++) {
    for (let j = 0; j < array.length - i - 1; j++) {
      // Highlight comparing bars
      bars[j].material.color.set(0xff6a6a);
      bars[j + 1].material.color.set(0xff6a6a);

      await delay(20);  // Slow down for visibility

      if (array[j] > array[j + 1]) {
        // Swap
        [array[j], array[j + 1]] = [array[j + 1], array[j]];

        // Animate swap
        await animateSwap(bars[j], bars[j + 1]);
        [bars[j], bars[j + 1]] = [bars[j + 1], bars[j]];
      }

      // Reset color
      bars[j].material.color.set(0x4a9eff);
      bars[j + 1].material.color.set(0x4a9eff);
    }

    // Mark sorted (green)
    bars[array.length - i - 1].material.color.set(0x6aff6a);
  }
}

function delay(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

async function animateSwap(bar1, bar2) {
  const pos1 = bar1.position.x;
  const pos2 = bar2.position.x;

  const steps = 10;
  for (let i = 0; i <= steps; i++) {
    const t = i / steps;
    bar1.position.x = pos1 + (pos2 - pos1) * t;
    bar2.position.x = pos2 + (pos1 - pos2) * t;
    await delay(10);
  }
}

bubbleSort();
```

## Molecule Viewer

Interactive 3D molecular structures:

```javascript
// Water molecule (H2O)
const molecule = {
  atoms: [
    { element: 'O', position: [0, 0, 0], color: 0xff0000 },      // Oxygen (red)
    { element: 'H', position: [-0.76, 0.59, 0], color: 0xffffff }, // Hydrogen (white)
    { element: 'H', position: [0.76, 0.59, 0], color: 0xffffff },  // Hydrogen (white)
  ],
  bonds: [
    { from: 0, to: 1 },  // O-H
    { from: 0, to: 2 },  // O-H
  ],
};

// Render atoms
const atomSizes = { H: 0.4, O: 0.6, C: 0.7, N: 0.65 };

molecule.atoms.forEach((atom, i) => {
  const geometry = new THREE.SphereGeometry(atomSizes[atom.element], 32, 32);
  const material = new THREE.MeshStandardMaterial({
    color: atom.color,
    metalness: 0.3,
    roughness: 0.4,
  });

  const sphere = new THREE.Mesh(geometry, material);
  sphere.position.set(...atom.position);
  scene.add(sphere);
});

// Render bonds
molecule.bonds.forEach(({ from, to }) => {
  const start = new THREE.Vector3(...molecule.atoms[from].position);
  const end = new THREE.Vector3(...molecule.atoms[to].position);

  const direction = new THREE.Vector3().subVectors(end, start);
  const length = direction.length();

  const geometry = new THREE.CylinderGeometry(0.08, 0.08, length, 8);
  const material = new THREE.MeshStandardMaterial({ color: 0x999999 });

  const bond = new THREE.Mesh(geometry, material);
  bond.position.copy(start).add(direction.multiplyScalar(0.5));
  bond.quaternion.setFromUnitVectors(
    new THREE.Vector3(0, 1, 0),
    direction.normalize()
  );

  scene.add(bond);
});

// Auto-rotate molecule
function animate() {
  requestAnimationFrame(animate);
  scene.rotation.y += 0.005;
  controls.update();
  renderer.render(scene, camera);
}
```

**Load molecules from PDB files:**

```javascript
async function loadMolecule(pdbUrl) {
  const response = await fetch(pdbUrl);
  const pdbText = await response.text();

  const atoms = [];
  const lines = pdbText.split('\n');

  for (const line of lines) {
    if (line.startsWith('ATOM')) {
      const element = line.slice(76, 78).trim();
      const x = parseFloat(line.slice(30, 38));
      const y = parseFloat(line.slice(38, 46));
      const z = parseFloat(line.slice(46, 54));

      atoms.push({ element, position: [x, y, z] });
    }
  }

  return atoms;
}

// Fetch protein structure from RCSB PDB
const atoms = await loadMolecule('https://files.rcsb.org/download/1MBN.pdb');
```

## Network Graph (Lesson Dependencies)

Visualize lesson prerequisites as force-directed graph:

```javascript
// Lesson dependency data
const lessons = [
  { id: 'bun-basics', title: 'Bun Basics', prerequisites: [] },
  { id: 'vanjs-intro', title: 'VanJS Intro', prerequisites: ['bun-basics'] },
  { id: 'routing', title: 'Routing', prerequisites: ['vanjs-intro'] },
  { id: 'htmx', title: 'HTMX', prerequisites: ['bun-basics'] },
  { id: 'auth', title: 'Authentication', prerequisites: ['routing', 'htmx'] },
];

// Create nodes
const nodes = lessons.map(lesson => {
  const geometry = new THREE.SphereGeometry(0.3, 32, 32);
  const material = new THREE.MeshStandardMaterial({
    color: 0x4a9eff,
    metalness: 0.5,
    roughness: 0.5,
  });

  const node = new THREE.Mesh(geometry, material);
  node.position.set(
    Math.random() * 10 - 5,
    Math.random() * 10 - 5,
    Math.random() * 10 - 5
  );

  node.userData = { id: lesson.id, title: lesson.title };
  scene.add(node);

  return { lesson, mesh: node };
});

// Create edges
lessons.forEach(lesson => {
  lesson.prerequisites.forEach(prereqId => {
    const from = nodes.find(n => n.lesson.id === prereqId);
    const to = nodes.find(n => n.lesson.id === lesson.id);

    if (from && to) {
      const points = [from.mesh.position, to.mesh.position];
      const geometry = new THREE.BufferGeometry().setFromPoints(points);
      const material = new THREE.LineBasicMaterial({ color: 0x666666 });
      const line = new THREE.Line(geometry, material);
      scene.add(line);
    }
  });
});

// Force-directed layout (simple spring simulation)
function updateForces() {
  const k = 0.01;  // Spring constant

  // Repulsion between nodes
  for (let i = 0; i < nodes.length; i++) {
    for (let j = i + 1; j < nodes.length; j++) {
      const dir = new THREE.Vector3()
        .subVectors(nodes[i].mesh.position, nodes[j].mesh.position);
      const dist = dir.length();

      if (dist < 5) {
        const force = k * (5 - dist);
        nodes[i].mesh.position.add(dir.normalize().multiplyScalar(force));
        nodes[j].mesh.position.sub(dir.normalize().multiplyScalar(force));
      }
    }
  }

  // Attraction along edges
  lessons.forEach(lesson => {
    lesson.prerequisites.forEach(prereqId => {
      const from = nodes.find(n => n.lesson.id === prereqId);
      const to = nodes.find(n => n.lesson.id === lesson.id);

      if (from && to) {
        const dir = new THREE.Vector3()
          .subVectors(to.mesh.position, from.mesh.position);
        const dist = dir.length();

        const force = k * (dist - 2);  // Target distance: 2 units
        to.mesh.position.sub(dir.normalize().multiplyScalar(force));
        from.mesh.position.add(dir.normalize().multiplyScalar(force));
      }
    });
  });
}

// Animate with force updates
let iterations = 0;
function animate() {
  requestAnimationFrame(animate);

  if (iterations < 300) {  // Run simulation for 300 frames
    updateForces();
    iterations++;
  }

  controls.update();
  renderer.render(scene, camera);
}
```

## Particle System (10K Particles)

GPU-accelerated particle rendering:

```javascript
const particleCount = 10000;
const positions = new Float32Array(particleCount * 3);
const colors = new Float32Array(particleCount * 3);
const velocities = new Float32Array(particleCount * 3);

for (let i = 0; i < particleCount; i++) {
  positions[i * 3] = Math.random() * 20 - 10;      // X
  positions[i * 3 + 1] = Math.random() * 20 - 10;  // Y
  positions[i * 3 + 2] = Math.random() * 20 - 10;  // Z

  velocities[i * 3] = (Math.random() - 0.5) * 0.1;
  velocities[i * 3 + 1] = (Math.random() - 0.5) * 0.1;
  velocities[i * 3 + 2] = (Math.random() - 0.5) * 0.1;

  const color = new THREE.Color().setHSL(Math.random(), 0.7, 0.6);
  colors[i * 3] = color.r;
  colors[i * 3 + 1] = color.g;
  colors[i * 3 + 2] = color.b;
}

const geometry = new THREE.BufferGeometry();
geometry.setAttribute('position', new THREE.BufferAttribute(positions, 3));
geometry.setAttribute('color', new THREE.BufferAttribute(colors, 3));

const material = new THREE.PointsMaterial({
  size: 0.1,
  vertexColors: true,
  transparent: true,
  opacity: 0.8,
});

const particles = new THREE.Points(geometry, material);
scene.add(particles);

// Animate particles
function animate() {
  requestAnimationFrame(animate);

  const positions = particles.geometry.attributes.position.array;

  for (let i = 0; i < particleCount; i++) {
    positions[i * 3] += velocities[i * 3];
    positions[i * 3 + 1] += velocities[i * 3 + 1];
    positions[i * 3 + 2] += velocities[i * 3 + 2];

    // Bounce at boundaries
    if (Math.abs(positions[i * 3]) > 10) velocities[i * 3] *= -1;
    if (Math.abs(positions[i * 3 + 1]) > 10) velocities[i * 3 + 1] *= -1;
    if (Math.abs(positions[i * 3 + 2]) > 10) velocities[i * 3 + 2] *= -1;
  }

  particles.geometry.attributes.position.needsUpdate = true;

  controls.update();
  renderer.render(scene, camera);
}
```

**Measured performance**: 60fps with 10,000 particles on integrated GPU.

## Integration with VanJS

Embed Three.js in reactive components:

```javascript
// client/components/Chart3D.js
import van from 'vanjs-core';
import * as THREE from 'three';

const { div } = van.tags;

export function Chart3D(data) {
  const container = div({ style: 'width: 800px; height: 600px;' });

  van.derive(() => {
    const scene = new THREE.Scene();
    const camera = new THREE.PerspectiveCamera(75, 800/600, 0.1, 1000);
    const renderer = new THREE.WebGLRenderer();

    renderer.setSize(800, 600);
    container.appendChild(renderer.domElement);

    // Create visualization from data.val
    createChart(scene, data.val);

    camera.position.set(5, 5, 5);

    function animate() {
      requestAnimationFrame(animate);
      renderer.render(scene, camera);
    }
    animate();
  });

  return container;
}

// Usage:
const stats = van.state([]);

fetch('/api/stats').then(r => r.json()).then(data => {
  stats.val = data;
});

van.add(document.body, Chart3D(stats));
```

## Common Pitfalls

### Memory Leaks

```javascript
// BAD - renderer never disposed
function createChart() {
  const renderer = new THREE.WebGLRenderer();
  // ... render chart
}

// GOOD - dispose on cleanup
function createChart() {
  const renderer = new THREE.WebGLRenderer();

  window.addEventListener('beforeunload', () => {
    renderer.dispose();
    scene.traverse(obj => {
      if (obj.geometry) obj.geometry.dispose();
      if (obj.material) obj.material.dispose();
    });
  });
}
```

### Poor Performance with Large Scenes

```javascript
// BAD - individual meshes for 10K objects
for (let i = 0; i < 10000; i++) {
  const cube = new THREE.Mesh(geometry, material);
  scene.add(cube);
}

// GOOD - instanced mesh
const geometry = new THREE.BoxGeometry();
const material = new THREE.MeshStandardMaterial();
const mesh = new THREE.InstancedMesh(geometry, material, 10000);

for (let i = 0; i < 10000; i++) {
  const matrix = new THREE.Matrix4();
  matrix.setPosition(Math.random() * 100, Math.random() * 100, Math.random() * 100);
  mesh.setMatrixAt(i, matrix);
}

scene.add(mesh);
```

**Instanced rendering**: 1 draw call for 10K objects instead of 10K draw calls. 100-1000x faster.
