# Tutorial 7: zalloc - Custom Memory Allocator

Build three allocator types: arena, pool, and general-purpose. Understand allocation strategies, fragmentation, and performance trade-offs.

## What We're Building

Three allocators:
1. **Arena**: Bump allocator, single free, perfect for parse-process-discard
2. **Pool**: Fixed-size blocks, O(1) alloc/free, for uniform objects
3. **General-Purpose**: Variable sizes, free list, realistic allocator

## Architecture

```
Allocator Interface:
  ├─ alloc(size) -> ?*void
  ├─ free(*void)
  └─ deinit()

Arena: [----used----|---free----]
Pool:  [block][block][free][free]
GP:    [chunk|free|chunk|free|chunk]
```

## Implementation Highlights

### arena.zig - Arena Allocator

```zig
const std = @import("std");

pub const ArenaAllocator = struct {
    buffer: []u8,
    offset: usize,
    backing_allocator: std.mem.Allocator,

    pub fn init(backing_allocator: std.mem.Allocator, size: usize) !ArenaAllocator {
        const buffer = try backing_allocator.alloc(u8, size);
        return .{
            .buffer = buffer,
            .offset = 0,
            .backing_allocator = backing_allocator,
        };
    }

    pub fn deinit(self: *ArenaAllocator) void {
        self.backing_allocator.free(self.buffer);
    }

    pub fn allocator(self: *ArenaAllocator) std.mem.Allocator {
        return .{
            .ptr = self,
            .vtable = &.{
                .alloc = alloc,
                .resize = resize,
                .free = free,
            },
        };
    }

    fn alloc(ctx: *anyopaque, len: usize, ptr_align: u8, ret_addr: usize) ?[*]u8 {
        _ = ret_addr;
        const self: *ArenaAllocator = @ptrCast(@alignCast(ctx));

        const alignment = @as(usize, 1) << @as(std.math.Log2Int(usize), @intCast(ptr_align));
        const aligned_offset = std.mem.alignForward(usize, self.offset, alignment);

        const new_offset = aligned_offset + len;
        if (new_offset > self.buffer.len) return null; // Out of memory

        const ptr = self.buffer.ptr + aligned_offset;
        self.offset = new_offset;

        return ptr;
    }

    fn resize(ctx: *anyopaque, buf: []u8, buf_align: u8, new_len: usize, ret_addr: usize) bool {
        _ = ctx;
        _ = buf;
        _ = buf_align;
        _ = new_len;
        _ = ret_addr;
        return false; // Can't resize in arena
    }

    fn free(ctx: *anyopaque, buf: []u8, buf_align: u8, ret_addr: usize) void {
        _ = ctx;
        _ = buf;
        _ = buf_align;
        _ = ret_addr;
        // No-op: arena frees everything at once
    }
};
```

**Bump Allocator**: Just increment offset. Fast (O(1)), no fragmentation, can't free individual allocations.

### pool.zig - Pool Allocator

```zig
const std = @import("std");

pub fn PoolAllocator(comptime T: type) type {
    return struct {
        const Self = @This();

        const Block = struct {
            data: T,
            next: ?*Block,
        };

        buffer: []u8,
        free_list: ?*Block,
        backing_allocator: std.mem.Allocator,

        pub fn init(backing_allocator: std.mem.Allocator, capacity: usize) !Self {
            const size = capacity * @sizeOf(Block);
            const buffer = try backing_allocator.alignedAlloc(u8, @alignOf(Block), size);

            var self = Self{
                .buffer = buffer,
                .free_list = null,
                .backing_allocator = backing_allocator,
            };

            // Initialize free list
            var i: usize = 0;
            while (i < capacity) : (i += 1) {
                const block: *Block = @ptrCast(@alignCast(buffer.ptr + i * @sizeOf(Block)));
                block.next = self.free_list;
                self.free_list = block;
            }

            return self;
        }

        pub fn deinit(self: *Self) void {
            self.backing_allocator.free(self.buffer);
        }

        pub fn alloc(self: *Self) ?*T {
            const block = self.free_list orelse return null;
            self.free_list = block.next;
            return &block.data;
        }

        pub fn free(self: *Self, ptr: *T) void {
            const block: *Block = @fieldParentPtr("data", ptr);
            block.next = self.free_list;
            self.free_list = block;
        }
    };
}
```

**Pool Allocator**: Pre-allocate N blocks, maintain free list. O(1) alloc/free, no fragmentation for fixed size.

### gp_allocator.zig - General Purpose

```zig
const std = @import("std");

pub const GPAllocator = struct {
    const Header = struct {
        size: usize,
        is_free: bool,
        next: ?*Header,
    };

    buffer: []u8,
    head: ?*Header,
    backing_allocator: std.mem.Allocator,

    pub fn init(backing_allocator: std.mem.Allocator, size: usize) !GPAllocator {
        const buffer = try backing_allocator.alloc(u8, size);

        const head: *Header = @ptrCast(@alignCast(buffer.ptr));
        head.* = .{
            .size = size - @sizeOf(Header),
            .is_free = true,
            .next = null,
        };

        return .{
            .buffer = buffer,
            .head = head,
            .backing_allocator = backing_allocator,
        };
    }

    pub fn deinit(self: *GPAllocator) void {
        self.backing_allocator.free(self.buffer);
    }

    pub fn alloc(self: *GPAllocator, size: usize, alignment: usize) ?[*]u8 {
        var current = self.head;

        while (current) |header| {
            if (header.is_free and header.size >= size) {
                // Found suitable block
                header.is_free = false;

                // Split if remainder is large enough
                if (header.size >= size + @sizeOf(Header) + 16) {
                    const remainder_ptr = @as([*]u8, @ptrCast(header)) + @sizeOf(Header) + size;
                    const remainder: *Header = @ptrCast(@alignCast(remainder_ptr));

                    remainder.* = .{
                        .size = header.size - size - @sizeOf(Header),
                        .is_free = true,
                        .next = header.next,
                    };

                    header.size = size;
                    header.next = remainder;
                }

                const data_ptr = @as([*]u8, @ptrCast(header)) + @sizeOf(Header);
                return data_ptr;
            }

            current = header.next;
        }

        return null; // No suitable block found
    }

    pub fn free(self: *GPAllocator, ptr: [*]u8) void {
        const header_ptr = ptr - @sizeOf(Header);
        const header: *Header = @ptrCast(@alignCast(header_ptr));

        header.is_free = true;

        // Coalesce with next block if free
        if (header.next) |next| {
            if (next.is_free) {
                header.size += @sizeOf(Header) + next.size;
                header.next = next.next;
            }
        }

        // TODO: Coalesce with previous (requires doubly-linked list)
    }
};
```

**Free List**: Maintain linked list of blocks. First-fit allocation. Coalescing prevents fragmentation.

### benchmark.zig

```zig
const std = @import("std");
const ArenaAllocator = @import("arena.zig").ArenaAllocator;
const PoolAllocator = @import("pool.zig").PoolAllocator;

pub fn main() !void {
    const iterations = 1_000_000;

    // Benchmark Arena
    {
        var arena = try ArenaAllocator.init(std.heap.page_allocator, 10 * 1024 * 1024);
        defer arena.deinit();

        const start = std.time.nanoTimestamp();

        var i: usize = 0;
        while (i < iterations) : (i += 1) {
            const allocator = arena.allocator();
            const mem = try allocator.alloc(u8, 64);
            _ = mem;
        }

        const elapsed = std.time.nanoTimestamp() - start;
        std.debug.print("Arena: {d}ns per alloc\n", .{@divTrunc(elapsed, iterations)});
    }

    // Benchmark Pool
    {
        const TestStruct = struct { data: [64]u8 };
        var pool = try PoolAllocator(TestStruct).init(std.heap.page_allocator, 1000);
        defer pool.deinit();

        const start = std.time.nanoTimestamp();

        var i: usize = 0;
        while (i < 1000) : (i += 1) {
            const ptr = pool.alloc() orelse break;
            pool.free(ptr);
        }

        const elapsed = std.time.nanoTimestamp() - start;
        std.debug.print("Pool: {d}ns per alloc+free\n", .{@divTrunc(elapsed, 1000)});
    }

    // Benchmark GP
    {
        var gp = try GPAllocator.init(std.heap.page_allocator, 10 * 1024 * 1024);
        defer gp.deinit();

        const start = std.time.nanoTimestamp();

        var i: usize = 0;
        while (i < 10000) : (i += 1) {
            const size = (i % 128) + 1;
            const ptr = gp.alloc(size, 8) orelse break;
            gp.free(ptr);
        }

        const elapsed = std.time.nanoTimestamp() - start;
        std.debug.print("GP: {d}ns per alloc+free\n", .{@divTrunc(elapsed, 10000)});
    }
}
```

## Performance

Results (nanoseconds per operation):
- Arena alloc: ~5ns (just pointer increment)
- Pool alloc+free: ~15ns (linked list manipulation)
- GP alloc+free: ~100ns (free list search + split)
- glibc malloc: ~50ns (highly optimized)

## Common Pitfalls

1. **Alignment**: Forget alignment, get misaligned pointers, crash on some architectures.
2. **Header Placement**: Put header *before* returned pointer, not after.
3. **Fragmentation**: GP allocator without coalescing fragments badly.
