# Tutorial 1: zgrep - Text Search Tool

Build a grep-like tool from scratch. File I/O, pattern matching, argument parsing, and proper resource management.

## What We're Building

```bash
$ zgrep "error" /var/log/syslog
Line 42: error: failed to connect
Line 108: error: timeout exceeded

$ zgrep -i "ERROR" *.log  # Case-insensitive
$ zgrep -n "TODO" src/**/*.zig  # Line numbers
```

Features: basic pattern matching (literal strings, no regex yet), recursive directory search, case-insensitive option, line numbers, and handles large files without loading entire file into memory.

## Architecture

```
main()
  ├─ parseArgs() -> Config
  ├─ searchPaths() -> []Match
  │   ├─ walkDirectory() (if dir)
  │   └─ searchFile()
  │       ├─ readLineByLine()
  │       └─ matchPattern()
  └─ printResults()
```

Key decision: stream processing. Don't load entire file - read line by line. Saves memory, handles arbitrarily large files.

pub fn build(b: *std.Build) void {
    const target = b.standardTargetOptions(.{});
    const optimize = b.standardOptimizeOption(.{});

    const exe = b.addExecutable(.{
        .name = "zgrep",
        .root_source_file = b.path("src/main.zig"),
        .target = target,
        .optimize = optimize,
    });

    b.installArtifact(exe);

    const run_cmd = b.addRunArtifact(exe);
    run_cmd.step.dependOn(b.getInstallStep());

    if (b.args) |args| {
        run_cmd.addArgs(args);
    }

    const run_step = b.step("run", "Run the app");
    run_step.dependOn(&run_cmd.step);
}

### args.zig - Command Line Parsing

```zig
const std = @import("std");

pub const Config = struct {
    pattern: []const u8,
    paths: []const []const u8,
    case_insensitive: bool = false,
    show_line_numbers: bool = false,
    recursive: bool = false,
};

pub fn parseArgs(allocator: std.mem.Allocator) !Config {
    var args = try std.process.argsWithAllocator(allocator);
    defer args.deinit();

    _ = args.skip(); // Skip program name

    var config: Config = undefined;
    var paths = std.ArrayList([]const u8).init(allocator);
    defer paths.deinit();

    var case_insensitive = false;
    var show_line_numbers = false;
    var recursive = false;

    // Parse flags
    while (args.next()) |arg| {
        if (std.mem.eql(u8, arg, "-i")) {
            case_insensitive = true;
        } else if (std.mem.eql(u8, arg, "-n")) {
            show_line_numbers = true;
        } else if (std.mem.eql(u8, arg, "-r")) {
            recursive = true;
        } else {
            // First non-flag is pattern
            if (config.pattern.len == 0) {
                config.pattern = arg;
            } else {
                try paths.append(arg);
            }
        }
    }

    if (config.pattern.len == 0) {
        return error.NoPattern;
    }

    // Default to stdin if no paths
    if (paths.items.len == 0) {
        try paths.append("-");
    }

    config.paths = try paths.toOwnedSlice();
    config.case_insensitive = case_insensitive;
    config.show_line_numbers = show_line_numbers;
    config.recursive = recursive;

    return config;
}
```

**Key Points**:
- `std.process.argsWithAllocator()`: Gets command line args, needs allocator for Windows compat
- Manual parsing: No clap or similar. Keep it simple.
- `ArrayList`: Dynamic array, like Python list. `toOwnedSlice()` transfers ownership.

### search.zig - Core Search Logic

```zig
const std = @import("std");

pub const Match = struct {
    filename: []const u8,
    line_number: usize,
    line: []const u8,
};

pub fn searchFile(
    allocator: std.mem.Allocator,
    filepath: []const u8,
    pattern: []const u8,
    case_insensitive: bool,
) !std.ArrayList(Match) {
    var matches = std.ArrayList(Match).init(allocator);
    errdefer matches.deinit();

    const file = try std.fs.cwd().openFile(filepath, .{});
    defer file.close();

    var buf_reader = std.io.bufferedReader(file.reader());
    var reader = buf_reader.reader();

    var line_buffer = std.ArrayList(u8).init(allocator);
    defer line_buffer.deinit();

    var line_num: usize = 1;
    while (true) {
        line_buffer.clearRetainingCapacity();
        reader.streamUntilDelimiter(line_buffer.writer(), '\n', null) catch |err| switch (err) {
            error.EndOfStream => break,
            else => return err,
        };

        if (matchesPattern(line_buffer.items, pattern, case_insensitive)) {
            const line_copy = try allocator.dupe(u8, line_buffer.items);
            try matches.append(.{
                .filename = filepath,
                .line_number = line_num,
                .line = line_copy,
            });
        }

        line_num += 1;
    }

    return matches;
}

fn matchesPattern(haystack: []const u8, needle: []const u8, case_insensitive: bool) bool {
    if (case_insensitive) {
        return std.ascii.indexOfIgnoreCase(haystack, needle) != null;
    } else {
        return std.mem.indexOf(u8, haystack, needle) != null;
    }
}

pub fn searchDirectory(
    allocator: std.mem.Allocator,
    dir_path: []const u8,
    pattern: []const u8,
    case_insensitive: bool,
    recursive: bool,
) !std.ArrayList(Match) {
    var matches = std.ArrayList(Match).init(allocator);
    errdefer matches.deinit();

    var dir = try std.fs.cwd().openDir(dir_path, .{ .iterate = true });
    defer dir.close();

    var iter = dir.iterate();
    while (try iter.next()) |entry| {
        if (entry.kind == .file) {
            const full_path = try std.fs.path.join(allocator, &.{ dir_path, entry.name });
            defer allocator.free(full_path);

            const file_matches = try searchFile(allocator, full_path, pattern, case_insensitive);
            defer file_matches.deinit();

            try matches.appendSlice(file_matches.items);
        } else if (entry.kind == .directory and recursive) {
            const subdir_path = try std.fs.path.join(allocator, &.{ dir_path, entry.name });
            defer allocator.free(subdir_path);

            const subdir_matches = try searchDirectory(allocator, subdir_path, pattern, case_insensitive, recursive);
            defer subdir_matches.deinit();

            try matches.appendSlice(subdir_matches.items);
        }
    }

    return matches;
}
```

**Implementation Notes**:

1. **Buffered Reading**: `std.io.bufferedReader()` wraps file reader. Reduces syscalls by reading chunks.

2. **Line-by-Line**: `streamUntilDelimiter()` reads until newline. No loading entire file.

3. **Error Handling**: `errdefer` runs cleanup on error return. Like try/finally but less verbose.

4. **Memory Management**: `allocator.dupe()` copies string. Necessary because `line_buffer` gets reused.

5. **Directory Walking**: `dir.iterate()` gives you entries one at a time. Recursive search calls itself.

### main.zig

```zig
const std = @import("std");
const args = @import("args.zig");
const search = @import("search.zig");

pub fn main() !void {
    var gpa = std.heap.GeneralPurposeAllocator(.{}){};
    defer _ = gpa.deinit();

    const allocator = gpa.allocator();

    const config = try args.parseArgs(allocator);
    defer allocator.free(config.paths);

    const stdout = std.io.getStdOut().writer();

    for (config.paths) |path| {
        const is_dir = blk: {
            const stat = std.fs.cwd().statFile(path) catch break :blk false;
            break :blk stat.kind == .directory;
        };

        const matches = if (is_dir)
            try search.searchDirectory(allocator, path, config.pattern, config.case_insensitive, config.recursive)
        else
            try search.searchFile(allocator, path, config.pattern, config.case_insensitive);

        defer {
            for (matches.items) |match| {
                allocator.free(match.line);
            }
            matches.deinit();
        }

        for (matches.items) |match| {
            if (config.show_line_numbers) {
                try stdout.print("{s}:{d}: {s}\n", .{ match.filename, match.line_number, match.line });
            } else {
                try stdout.print("{s}: {s}\n", .{ match.filename, match.line });
            }
        }
    }
}
```

**Main Function Notes**:
- `GeneralPurposeAllocator`: Detects memory leaks in debug mode. Production: use ArenaAllocator for short-lived programs.
- `statFile()`: Gets file metadata. Checks if path is directory.
- `blk: break :blk`: Labeled block expression. Returns value from block.
- Cleanup: Free allocated strings, then free ArrayList.

## Common Pitfalls

### 1. Memory Leaks

```zig
// Wrong - leaks line_copy
if (matchesPattern(line, pattern, false)) {
    const line_copy = try allocator.dupe(u8, line);
    try matches.append(.{ .line = line_copy });
}
// If append fails, line_copy is leaked

// Right - use errdefer
const line_copy = try allocator.dupe(u8, line);
errdefer allocator.free(line_copy);
try matches.append(.{ .line = line_copy });
```

### 2. Buffer Reuse

```zig
// Wrong - stores pointer to reused buffer
while (...) {
    line_buffer.clearRetainingCapacity();
    // Read into line_buffer
    try matches.append(.{ .line = line_buffer.items }); // All point to same buffer!
}

// Right - duplicate the data
const line_copy = try allocator.dupe(u8, line_buffer.items);
```

### 3. Path Joining

```zig
// Wrong - manual concatenation
const path = try std.fmt.allocPrint(allocator, "{s}/{s}", .{dir, file});

// Right - handles edge cases (trailing slashes, etc.)
const path = try std.fs.path.join(allocator, &.{dir, file});
```

### 4. EOF Handling

```zig
// Wrong - treats EOF as error
while (try reader.readUntilDelimiterOrEof(&buffer, '\n')) |line| {
    // Process
}

// Right - EOF is normal termination
while (reader.readUntilDelimiterOrEof(&buffer, '\n') catch |err| {
    if (err == error.EndOfStream) break;
    return err;
}) |line| {
    // Process
}
```

## Performance Analysis

**Memory usage**:
- Line buffer: ~4KB (reused)
- Match storage: O(matches × avg_line_length)
- Total: Minimal, scales with output not input

**Syscalls per 1GB file**:
- open: 1
- read: ~250 (4KB buffer, 1GB / 4KB = 256K but buffered reader chunks)
- close: 1

Python equivalent makes millions of allocations (one per line). We reuse buffers.

## Extensions

1. **Regex Support**: Use `std.regex` or write simple glob matcher
2. **Binary File Detection**: Skip binary files (check for null bytes in first chunk)
3. **Context Lines**: Show N lines before/after match (`-A`, `-B`, `-C` flags)
4. **Parallel Search**: Use thread pool for multiple files
5. **Memory Mapping**: `std.fs.File.mmap()` for huge files
